/** All simulation parameters in SI units
 * 
 * UNITS USED THROUGHOUT:
 * - Time: seconds (s)
 * - Length: meters (m)
 * - Amount: moles (mol)
 * - Concentration: mol/m³
 * - Pressure: Pascals (Pa)
 * - Mass: kilograms (kg)
 * - Energy: Joules (J)
 * 
 * NO UNIT CONVERSIONS IN CODE!
 */
public class SimParams {
    
    // =================================================================
    // DOMAIN & DISCRETIZATION
    // =================================================================
    public static final double CELL_LENGTH = 10e-6;  // m (10 μm)
    public static final int GRID_SIZE_CELLS = 400;   // cells
    public static final double DOMAIN_SIZE = GRID_SIZE_CELLS * CELL_LENGTH;  // m
    public static final double TIME_STEP = 3600.0;   // s (1 hour)
    
    // =================================================================
    // PHARMACOKINETICS
    // =================================================================
    public static final double LAMBDA_BIO = 1.6e-4 / 60.0;      // 1/s
    public static final double LAMBDA_DECAY = 7.14e-5 / 60.0;   // 1/s (Lu-177)
    public static final double K_ON = 0.046 / 60.0 / 1000.0;    // m³/(mol·s)
    public static final double K_OFF = 0.046 / 60.0;            // 1/s
    public static final double K_INT = 0.001 / 60.0;            // 1/s
    public static final double K_REL = 2e-4 / 60.0;             // 1/s
    
    public static final double V_CENTRAL = 0.458e-3;            // m³ (0.458 L)
    public static final double INTERSTITIAL_FRACTION = 0.4;     // dimensionless
    
    // Per vessel parameters
    public static final double VESSEL_VOLUME = 1e-6 * 1e-6 * 4000e-6;  // m³
    public static final double VESSEL_FLOW = 5e-9 / 1000.0;     // m³/s (5 nL/s)
    public static final double VESSEL_PS = 6.0 / 5.0 * VESSEL_FLOW;  // m³/s
    
    // =================================================================
    // CELL BIOLOGY
    // =================================================================
    public static final double RECEPTORS_PER_CELL = 5e5;        // count
    public static final double AVOGADRO = 6.022e23;             // 1/mol
    public static final double CELL_CYCLE_DURATION = 86400.0;   // s (1 day)
    
    // =================================================================
    // OXYGEN DIFFUSION
    // =================================================================
    public static final double D_O2 = 2e-5 * 1e-4;              // m²/s (2e-5 cm²/s)
    public static final double P_O2_VESSEL = 100.0 * 133.322;   // Pa (100 mmHg)
    public static final double P_O2_VEIN = 30.0 * 133.322;      // Pa (30 mmHg)
    public static final double HENRY_O2 = 1.3e-3 / 760.0 * 1e9 / 1e-15;  // mol/(m³·Pa)
    
    public static final double HYPOXIC_THRESHOLD = 10.0 * 133.322;   // Pa
    public static final double NECROTIC_THRESHOLD = 0.5 * 133.322;   // Pa
    
    // Consumption rates (mol/(s·cell))
    public static final double CONSUMPTION_HEALTHY = 500.0 * SOME_CONVERSION;
    public static final double CONSUMPTION_NORMAL = 6000.0 * SOME_CONVERSION;
    // ... etc
    
    // =================================================================
    // RADIOBIOLOGY
    // =================================================================
    public static final double ALPHA_NORMAL = 0.15;             // Gy⁻¹
    public static final double ALPHA_HYPOXIC = 0.0107;          // Gy⁻¹
    public static final double BETA_NORMAL = 0.048;             // Gy⁻²
    public static final double BETA_HYPOXIC = 0.0024;           // Gy⁻²
    public static final double REPAIR_RATE = 0.3 / 3600.0;      // 1/s (0.3/hr)
    
    public static final double E_BETA = 2.14e-14;               // J (Lu-177 beta)
    
    // =================================================================
    // INJECTION PROTOCOL (defaults)
    // =================================================================
    public static final double DOSE_PER_INJECTION = 100e-9;     // mol (100 nmol)
    public static final double HOT_FRACTION = 0.1;              // dimensionless
    public static final int[] INJECTION_DAYS = {5, 35, 65, 95}; // days
    
    // =================================================================
    // CELL TYPES
    // =================================================================
    public static final int HEALTHY = 0;
    public static final int NORMAL = 1;
    public static final int HYPOXIC = 2;
    public static final int NECROTIC = 3;
    public static final int APOPTOTIC = 4;
    public static final int VESSEL = 5;
    
    // =================================================================
    // COMPUTED PARAMETERS (derived from above)
    // =================================================================
    public static double computeTumorVolume(int cellCount) {
        return cellCount * Math.pow(CELL_LENGTH, 3);
    }
    
    public static double computeReceptorMoles(int cellCount) {
        return cellCount * RECEPTORS_PER_CELL / AVOGADRO;
    }
}
